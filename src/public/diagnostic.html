Student List Restorer

Creates a new student list directly in the page, bypassing all tab and visibility issues.

// Create a new student list
(function() {
  fetch('/api/students')
    .then(response => response.json())
    .then(students => {
      // Create container
      const container = document.createElement('div');
      container.style.cssText = 'padding: 20px; background: white; border-radius: 8px; margin: 20px 0;';
      
      // Create heading
      const heading = document.createElement('h2');
      heading.textContent = `Students (${students.length})`;
      container.appendChild(heading);
      
      // Create grid
      const grid = document.createElement('div');
      grid.style.cssText = 'display: grid; grid-template-columns: repeat(auto-fill, minmax(200px, 1fr)); gap: 20px;';
      container.appendChild(grid);
      
      // Add students
      students.slice(0, 20).forEach(student => {
        const card = document.createElement('div');
        card.style.cssText = 'border: 1px solid #eee; border-radius: 8px; overflow: hidden;';
        
        const img = document.createElement('img');
        img.src = student.imagePath || '/uploads/student-images/placeholder.png';
        img.style.cssText = 'width: 100%; height: 200px; object-fit: cover;';
        
        const info = document.createElement('div');
        info.style.cssText = 'padding: 10px;';
        info.innerText = student.fullName;
        
        card.appendChild(img);
        card.appendChild(info);
        grid.appendChild(card);
      });
      
      // Add to page
      document.body.insertBefore(container, document.body.firstChild);
    });
})();

Tab System Explorer

Analyzes the tab system and attempts to fix issues by directly controlling CSS.

// Analyze and fix tab system
(function() {
  console.log('TAB SYSTEM EXPLORER');
  
  const tabs = document.querySelectorAll('.tab-content');
  console.log(`Found ${tabs.length} tab elements`);
  
  tabs.forEach((tab, index) => {
    const id = tab.id || `tab-${index}`;
    const display = getComputedStyle(tab).display;
    const hasActiveClass = tab.classList.contains('active');
    
    console.log(`Tab #${index}: ${id} (display: ${display}, active: ${hasActiveClass})`);
    
    // Fix all tabs that should be active
    if (id === 'upload-tab') {
      console.log('  -> Setting upload-tab to display:block and adding active class');
      tab.style.setProperty('display', 'block', 'important');
      tab.classList.add('active');
      
      // Also fix the tab button
      const btn = document.getElementById('upload-tab-btn');
      if (btn) {
        btn.classList.add('active');
      }
      
      // Find student list and fix it
      const studentList = tab.querySelector('#student-list');
      if (studentList) {
        console.log('  -> Setting student-list to display:grid');
        studentList.style.setProperty('display', 'grid', 'important');
        studentList.style.setProperty('grid-template-columns', 'repeat(auto-fill, minmax(200px, 1fr))', 'important');
        studentList.style.setProperty('gap', '1rem', 'important');
      } else {
        console.log('  -> Could not find student-list inside tab');
      }
    }
  });
})();

Deep DOM Surgery

Performs deeper DOM manipulation by recreating the student list element.

// Deep DOM Surgery
(function() {
  console.log('DEEP DOM SURGERY');
  
  // Get upload tab element
  const uploadTab = document.getElementById('upload-tab');
  if (!uploadTab) {
    console.error('Upload tab not found');
    return;
  }
  
  // Force display
  uploadTab.style.setProperty('display', 'block', 'important');
  uploadTab.classList.add('active');
  
  // Get the student list
  const studentList = document.getElementById('student-list');
  if (!studentList) {
    console.error('Student list not found');
    return;
  }
  
  // Get current container
  const container = studentList.parentElement;
  
  // Create replacement student list
  const newStudentList = document.createElement('div');
  newStudentList.id = 'student-list-fixed';
  newStudentList.className = 'card-grid';
  newStudentList.style.cssText = 'display: grid !important; grid-template-columns: repeat(auto-fill, minmax(200px, 1fr)) !important; gap: 1.5rem !important;';
  
  // Add a loading message
  const loadingMsg = document.createElement('div');
  loadingMsg.style.cssText = 'grid-column: 1/-1; text-align: center; padding: 20px;';
  loadingMsg.textContent = 'Loading students...';
  newStudentList.appendChild(loadingMsg);
  
  // Insert new element
  container.insertBefore(newStudentList, studentList);
  
  // Fetch and display students
  fetch('/api/students')
    .then(response => response.json())
    .then(students => {
      // Clear loading message
      newStudentList.innerHTML = '';
      
      // Add students
      students.slice(0, 20).forEach(student => {
        const card = document.createElement('div');
        card.className = 'student-card';
        
        const img = document.createElement('img');
        img.className = 'student-img';
        img.src = student.imagePath || '/uploads/student-images/placeholder.png';
        
        const info = document.createElement('div');
        info.className = 'student-info';
        
        const name = document.createElement('div');
        name.className = 'student-name';
        name.textContent = student.fullName;
        
        info.appendChild(name);
        card.appendChild(img);
        card.appendChild(info);
        
        newStudentList.appendChild(card);
      });
      
      console.log(`Displayed ${students.length} students in new container`);
    })
    .catch(error => {
      console.error('Error fetching students:', error);
      newStudentList.innerHTML = '
Error loading students
';
    });
})();

CSS Rules Inspector

Analyzes all CSS rules affecting the student list element.

// CSS Rules Inspector
(function() {
  console.log('CSS RULES INSPECTOR');
  
  const studentList = document.getElementById('student-list');
  if (!studentList) {
    console.error('Student list element not found');
    return;
  }
  
  console.log('Student list element:', studentList);
  
  // Get all stylesheets
  const allRules = [];
  for (let i = 0; i < document.styleSheets.length; i++) {
    try {
      const sheet = document.styleSheets[i];
      const rules = sheet.cssRules || sheet.rules;
      
      for (let j = 0; j < rules.length; j++) {
        const rule = rules[j];
        
        // Check if this rule applies to the student list
        if (studentList.matches(rule.selectorText)) {
          allRules.push({
            selector: rule.selectorText,
            css: rule.style.cssText,
            sheet: sheet.href || 'inline'
          });
        }
      }
    } catch (e) {
      console.warn('Could not access stylesheet:', e);
    }
  }
  
  console.log('CSS Rules affecting student list:', allRules);
  
  // Show the computed style
  const style = window.getComputedStyle(studentList);
  console.log('Computed style for student list:');
  console.log('- display:', style.display);
  console.log('- visibility:', style.visibility);
  console.log('- opacity:', style.opacity);
  console.log('- position:', style.position);
  
  // Check parent elements
  let parent = studentList.parentElement;
  let depth = 0;
  
  while (parent && depth < 5) {
    const parentStyle = window.getComputedStyle(parent);
    console.log(`Parent (${parent.tagName}${parent.id ? '#'+parent.id : ''}${parent.className ? '.'+parent.className.replace(/ /g, '.') : ''}):`, {
      display: parentStyle.display,
      visibility: parentStyle.visibility,
      opacity: parentStyle.opacity,
      position: parentStyle.position
    });
    
    parent = parent.parentElement;
    depth++;
  }
})();
